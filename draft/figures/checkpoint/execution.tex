% An open arrowhead, with dimensions in PDF points.
\newcommand{\flowarrow}[4]{%
  \draw[line width=.75bp, line cap=round, line join=round]
    (#1) -- (#2);
  \draw[line width=.75bp, line cap=round, line join=round]
    ([xshift=#3bp,yshift=#4bp]#2) -- (#2)
    -- ([xshift=-#3bp,yshift=#4bp]#2);
}
